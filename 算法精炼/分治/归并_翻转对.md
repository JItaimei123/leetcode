给定一个数组 `nums`，如果 `i < j` 且 `nums[i] > 2*nums[j]` 我们就将 `(i, j)` 称作一个*重要翻转对*。

你需要返回给定数组中的重要翻转对的数量。

示例 1:

```C++
输入: [1,3,2,3,1]
输出: 2
```

思路：整体还是在用逆序对的思路，只不过这里是变形了，这里需要先根据条件统计`nums[i] > 2*nums[j]`统计一个阶段的ans，再进行合并再统计，原因是判断条件与归并排序原本的排序条件不一样，正常归并排序的运行条件只需要判断`nums[i]和nums[j]`的关系，而逆序对的统计和合并共享同一个指针移动逻辑，所以可以和排序一起进行，但这里两个条件并不相同，会造成冲突

```C++
class Solution 
{
    vector<int> tmp;
    int ans = 0;
public:
    int reversePairs(vector<int>& nums) 
    {
        tmp.resize(nums.size());
        mergesort(nums,0,nums.size()-1);
        return ans;    
    }
    void mergesort(vector<int>& nums,int left,int right)
    {
        if(left >= right) return;
        //1.确定分割元素
        int mid = left + (right-left)/2;
        //2.递归分割
        mergesort(nums,left,mid);
        mergesort(nums,mid+1,right);
        //3.先统计个数
        int cur1 = left,cur2 = mid+1,i = 0;
        while(cur1 <= mid && cur2 <= right)
        {
            if((long long) nums[cur1] > (long long) 2 * nums[cur2])
            {
            ans += mid - cur1 + 1;
            cur2++;
            }
            else cur1++;
        }
        //4.合并
        cur1 = left;
        cur2 = mid+1;
        while(cur1 <= mid && cur2 <= right)
        {
            if(nums[cur1] <= nums[cur2]) tmp[i++] = nums[cur1++];
            else tmp[i++] = nums[cur2++];
        }
        //处理未完成的情况
        while(cur1 <= mid) tmp[i++] = nums[cur1++];
        while(cur2 <= right) tmp[i++] = nums[cur2++];
        //5.还原数组
        for(int j = left;j<=right;j++)
        nums[j] = tmp[j-left];
    }
};
```
