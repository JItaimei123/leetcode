给你一个整数数组 `nums`。如果任一值在数组中出现 至少两次，返回 `true`；如果数组中每个元素互不相同，返回 `false`。

```C++
class Solution {
public:
    bool containsDuplicate(vector<int>& nums) 
    {
        unordered_map<int,int> hash;
        for(auto n : nums)
        {
        hash[n]++;
        if(hash[n] >= 2)
        return true;
        }
        return false;     
    }
};
```
